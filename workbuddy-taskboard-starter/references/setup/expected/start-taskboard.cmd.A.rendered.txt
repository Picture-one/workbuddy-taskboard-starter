@echo off
setlocal

set "APP_DIR=C:\Users\demo\.workbuddy\apps\dashi-taskboard"
set "CODEX_TASKBOARD_HOST=127.0.0.1"
set "CODEX_TASKBOARD_PORT=47823"
set "CODEX_TASKBOARD_DATA_DIR=C:\Users\demo\.workbuddy\taskboard-data"
